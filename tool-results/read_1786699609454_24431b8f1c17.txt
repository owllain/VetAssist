    80→    80→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    81→    81→  const ref = useRef<HTMLSpanElement>(null);
    82→    82→  const isInView = useInView(ref, { once: true });
    83→    83→  const [count, setCount] = useState(0);
    84→    84→
    85→    85→  useEffect(() => {
    86→    86→    if (!isInView) return;
    87→    87→    let start = 0;
    88→    88→    const end = target;
    89→    89→    const duration = 1500;
    90→    90→    const stepTime = 30;
    91→    91→    const steps = duration / stepTime;
    92→    92→    const increment = end / steps;
    93→    93→
    94→    94→    const timer = setInterval(() => {
    95→    95→      start += increment;
    96→    96→      if (start >= end) {
    97→    97→        setCount(end);
    98→    98→        clearInterval(timer);
    99→    99→      } else {
   100→   100→        setCount(Math.floor(start));
   101→   101→      }
   102→   102→    }, stepTime);
   103→   103→    return () => clearInterval(timer);
   104→   104→  }, [isInView, target]);
   105→   105→
   106→   106→  return <span ref={ref}>{count}{suffix}</span>;
   107→   107→}
