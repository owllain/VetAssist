    44→interface CalcResult {
    45→  medication: {
    46→    id: string;
    47→    name: string;
    48→    genericName: string;
    49→    category: string;
    50→    brandNames: string[];

    63→
    64→interface HistoryEntry {
    65→  type: 'medication' | 'free' | 'food';
    66→  timestamp: number;
    67→  summary: string;
    68→}
    69→
    70→const STORAGE_KEY = 'vetcalc-history';
    71→
    72→function loadHistory(): HistoryEntry[] {
    73→  try { return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]'); } catch { return []; }
    74→}
    75→
    76→function saveHistory(entry: HistoryEntry) {
    77→  const history = loadHistory();
    78→  history.unshift(entry);
    79→  if (history.length > 20) history.pop();
    80→  localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
    81→}
    82→
    83→function StepHeading({ num, children }: { num: number; children: React.ReactNode }) {
    84→  return (
    85→    <h3 className="text-base font-semibold mb-3 flex items-center gap-2.5">
    86→      <span className="step-number">{num}</span>
    87→      {children}
    88→    </h3>
    89→  );
    90→}

   171→          weight: parseFloat(weight),

   193→      setError('Error de conexión. Intente de nuevo.');
   194→    } finally {
   195→      setLoading(false);
   196→    }
   197→  };

   199→  const handleReset = () => {

   201→    setResult(null);
   202→    setError(null);
   203→  };

   218→                className="flex-shrink-0 bg-muted/60 rounded-lg px-3 py-1.5 text-xs text-muted-foreground border border-border/50"

   308→                selectedCategory === cat.id
   309→                  ? 'bg-primary text-primary-foreground border-primary shadow-md shadow-primary/20'

   313→              <span>{cat.icon}</span>

   316→          ))}
   317→        </div>
   318→      </div>
   319→
   320→      {/* Step 4: Medication List with Search */}
   321→      <AnimatePresence mode="wait">
   322→        {selectedCategory && filteredMedications.length > 0 && (
   323→          <motion.div
   324→            key={selectedCategory}
   325→            initial={{ opacity: 0, y: 10 }}
   326→            animate={{ opacity: 1, y: 0 }}
   327→            exit={{ opacity: 0, y: -10 }}
   328→            transition={{ duration: 0.2 }}
   329→          >
   330→            <div className="flex items-center justify-between gap-3 mb-3">
   331→              <StepHeading num={4}>Seleccione un Medicamento</StepHeading>

   334→                <Input
   335→                  type="text"
   336→                  placeholder="Buscar..."
   337→                  value={searchQuery}
   338→                  onChange={(e) => setSearchQuery(e.target.value)}
   339→                  className="h-8 text-sm pl-8"
   340→                />
   341→              </div>
   342→            </div>

   344→            {/* Dose Reference Table - collapsible */}
   345→            <Collapsible open={showRefTable} onOpenChange={setShowRefTable} className="mb-3 no-print">
   346→              <CollapsibleTrigger asChild>
   347→                <Button
   348→                  variant="ghost"
   349→                  size="sm"
   350→                  className="w-full justify-between gap-2 text-xs text-muted-foreground hover:text-foreground h-8 px-2"
   351→                >
   352→                  <span className="flex items-center gap-1.5">
   353→                    <ClipboardList size={14} weight="outline" />
   354→                    Ver tabla de referencia
   355→                  </span>
   356→                  <motion.span

   358→                    transition={{ duration: 0.2 }}
   359→                  >
   360→                    <ChevronDown size={14} weight="outline" />
   361→                  </motion.span>
   362→                </Button>
   363→              </CollapsibleTrigger>
   364→              <CollapsibleContent>
   365→                <motion.div
   366→                  initial={{ opacity: 0, height: 0 }}
   367→                  animate={{ opacity: 1, height: 'auto' }}
   368→                  exit={{ opacity: 0, height: 0 }}

   373→              </CollapsibleContent>
   374→            </Collapsible>
   375→
   376→            <div className="grid gap-2.5 max-h-[400px] overflow-y-auto pr-1">
   377→              {filteredMedications.map((med) => {
   378→                const isAvailable = med.species.includes(animalType);
   379→                const isSelected = selectedMedication?.id === med.id;
   380→                const medIsFav = isFavorite(med.id);
   381→
   382→                return (
   383→                  <Card
   384→                    key={med.id}
   385→                    className={`med-card cursor-pointer ${isSelected ? 'selected' : ''} ${
   386→                      !isAvailable ? 'opacity-40 pointer-events-none' : ''
   387→                    }`}
   388→                    onClick={() => {
   389→                      if (!isAvailable) return;
   390→                      setSelectedMedication(isSelected ? null : med);
   391→                      setResult(null);
   392→                    }}
   393→                  >
   394→                    <CardContent className="p-3.5">
   395→                      <div className="flex items-start justify-between gap-2">
   396→                        <div className="flex-1 min-w-0">
   397→                          <div className="flex items-center gap-2 flex-wrap">
   398→                            <h4 className="font-bold text-sm">{med.name}</h4>
   399→                            {med.species.map((s) => (
   400→                              <Badge
   401→                                key={s}
   402→                                variant={s === animalType ? 'default' : 'secondary'}
   403→                                className="text-[10px] px-1.5 py-0"
   404→                              >
   405→                                {s === 'perro' ? '🐕' : '🐈'} {s}
   406→                              </Badge>

   408→                          </div>
   409→                          <p className="text-xs text-muted-foreground mt-0.5">{med.genericName}</p>
   410→                          <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   411→                            <Badge variant="outline" className="text-[10px] font-semibold text-primary border-primary/30">

   413→                                ? `${med.doseMin} ${med.unit}`

   416→                            {med.route.map((r) => (
   417→                              <Badge key={r} variant="secondary" className="text-[10px] px-1.5 py-0">
   418→                                {r}

   420→                            ))}
   421→                          </div>

   423→                        <div className="flex items-center gap-1 flex-shrink-0 mt-1">
   424→                          <button
   425→                            onClick={(e) => {
   426→                              e.stopPropagation();
   427→                              toggleFav(med);

   430→                              medIsFav ? 'text-amber-500' : 'text-muted-foreground/30 hover:text-amber-400'
   431→                            }`}
   432→                            title={medIsFav ? 'Quitar de favoritos' : 'Agregar a favoritos'}
   433→                          >
   434→                            <Star size={16} weight={medIsFav ? 'fill' : 'outline'} />
   435→                          </button>
   436→                          <ChevronRight
   437→                            size={18} weight="outline"
   438→                            className={`transition-all ${isSelected ? 'text-primary rotate-90' : 'text-muted-foreground/40'}`}
   439→                          />
   440→                        </div>
   441→                      </div>
   442→                    </CardContent>
   443→                  </Card>

   445→              })}
   446→            </div>
   447→          </motion.div>
   448→        )}
   449→        {selectedCategory && filteredMedications.length === 0 && (
   450→          <motion.div
   451→            key="empty"
   452→            initial={{ opacity: 0 }} animate={{ opacity: 1 }}

   455→            <Search size={32} weight="outline" className="mx-auto mb-2 opacity-30" />
   456→            <p className="text-sm">No se encontraron medicamentos</p>

   459→      </AnimatePresence>
   460→
   461→      {/* Calculate Button */}
   462→      <AnimatePresence>

   464→          <motion.div

   466→            className="flex flex-col items-center gap-3 no-print"
   467→          >
   468→            <Button
   469→              size="lg"
   470→              onClick={handleCalculate}
   471→              disabled={!canCalculate || loading}
   472→              className="vet-pulse text-lg px-8 py-6 bg-primary hover:bg-primary/90 text-primary-foreground rounded-xl shadow-lg shadow-primary/20"

   476→                  <span className="animate-spin">⚙️</span>

   493→      {/* Error */}

   503→      </AnimatePresence>

   511→            exit={{ opacity: 0, y: -10, scale: 0.95 }}
   512→            transition={{ duration: 0.3 }}
