    80→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    81→  const ref = useRef<HTMLSpanElement>(null);
    82→  const isInView = useInView(ref, { once: true });
    83→  const [count, setCount] = useState(0);
    84→
    85→  useEffect(() => {
    86→    if (!isInView) return;
    87→    let start = 0;
    88→    const end = target;
    89→    const duration = 1500;
    90→    const stepTime = 30;
    91→    const steps = duration / stepTime;
    92→    const increment = end / steps;
    93→
    94→    const timer = setInterval(() => {
    95→      start += increment;
    96→      if (start >= end) {
    97→        setCount(end);
    98→        clearInterval(timer);
    99→      } else {
   100→        setCount(Math.floor(start));
   101→      }
   102→    }, stepTime);
   103→    return () => clearInterval(timer);
   104→  }, [isInView, target]);
   105→
   106→  return <span ref={ref}>{count}{suffix}</span>;
   107→}

   113→        {icon}
   114→      </div>
