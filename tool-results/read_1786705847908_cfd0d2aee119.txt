   108→/* ─── Weight-based maintenance lookup ───────────────────── */
   109→
   110→function getWeightBasedRate(weightKg: number): {
   111→  dailyMin: number;
   112→  dailyMax: number;
   113→  hourlyMin: number;
   114→  hourlyMax: number;
   115→  range: string;
   116→} {
   117→  if (weightKg <= 2) {
   118→    return {
   119→      dailyMin: 120, dailyMax: 150,
   120→      hourlyMin: 5, hourlyMax: 6,
   121→      range: '0–2 kg',
   122→    };
   123→  }
   124→  if (weightKg <= 5) {
   125→    return {
   126→      dailyMin: 80, dailyMax: 100,
   127→      hourlyMin: 3, hourlyMax: 4,
   128→      range: '2–5 kg',
   129→    };
   130→  }
   131→  if (weightKg <= 10) {
   132→    return {
   133→      dailyMin: 60, dailyMax: 80,
   134→      hourlyMin: 2.5, hourlyMax: 3.5,
   135→      range: '5–10 kg',
   136→    };
   137→  }
   138→  if (weightKg <= 20) {
   139→    return {
   140→      dailyMin: 50, dailyMax: 60,
   141→      hourlyMin: 2, hourlyMax: 2.5,
   142→      range: '10–20 kg',
   143→    };
   144→  }
   145→  // 20+ kg
   146→  return {
   147→    dailyMin: 40, dailyMax: 50,
   148→    hourlyMin: 1.5, hourlyMax: 2,
   149→    range: '>20 kg',
   150→  };
   151→}

   153→/* ─── Simplified formula ─────────────────────────────────── */

   155→function simplifiedFormula(weightKg: number): number {
   156→  return 30 * weightKg + 70;
   157→}

   170→/* ─── Visual Fluid Bag ────────────────────────────────────── */
   171→
   172→function FluidBagIndicator({ filled, total }: { filled: number; total: number }) {
   173→  const pct = Math.min((filled / total) * 100, 100);
   174→  return (
   175→    <div className="relative w-16 h-24 flex-shrink-0">
   176→      {/* Bag outline */}
   177→      <div className="absolute inset-0 rounded-lg border-2 border-primary/40 bg-primary/5 overflow-hidden">
   178→        {/* Fill level */}
   179→        <motion.div
   180→          className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-primary/60 to-primary/20"
   181→          initial={{ height: 0 }}
   182→          animate={{ height: `${pct}%` }}
   183→          transition={{ duration: 0.8, ease: 'easeOut' }}
   184→        />
   185→        {/* Top port */}
   186→        <div className="absolute top-0 left-1/2 -translate-x-1/2 w-4 h-2 bg-primary/30 rounded-b" />
   187→      </div>
   188→      {/* Label */}
   189→      <div className="absolute -bottom-5 left-1/2 -translate-x-1/2 text-[10px] font-semibold text-muted-foreground whitespace-nowrap">
   190→        {total}mL
   191→      </div>
   192→    </div>
   193→  );
   194→}

   196→/* ─── Infusion Time Card ─────────────────────────────────── */
   197→
   198→function InfusionTimeCard({
   199→  bagSize,
   200→  hours,
   201→  hourlyRate,
   202→}: {
   203→  bagSize: number;
   204→  hours: number;
   205→  hourlyRate: number;
   206→}) {
   207→  const h = Math.floor(hours);
   208→  const m = Math.round((hours - h) * 60);
   209→  const needsMultipleBags = hours < hourlyRate > 0 && bagSize / hourlyRate < hours;
   210→
   211→  return (
   212→    <div className="flex items-center gap-3 hover-tap rounded-lg p-2 border border-border/50">
   213→      <FluidBagIndicator filled={bagSize} total={bagSize} />
   214→      <div className="flex-1 min-w-0">
   215→        <div className="text-sm font-semibold">Bolsa {bagSize} mL</div>
   216→        <div className="text-xs text-muted-foreground">
   217→          {h > 0 ? `${h}h` : ''}{h > 0 && m > 0 ? ' ' : ''}{m > 0 ? `${m}min` : ''}
   218→          {h === 0 && m === 0 ? ' < 1 min' : ''}
   219→        </div>
   220→      </div>
   221→      <Badge variant="outline" className="text-[10px] font-mono flex-shrink-0">
   222→        {h}:{m.toString().padStart(2, '0')}
   223→      </Badge>
   224→    </div>
   225→  );
   226→}

   230→export default function IVFluidCalculator() {

   232→  const [species, setSpecies] = useState<Species>('perro');
   233→
   234→  // Step 2: Weight
   235→  const [weight, setWeight] = useState('');
   236→

   239→

   242→
   243→  // Advanced: Dehydration
   244→  const [dehydration, setDehydration] = useState<number>(0);
   245→
   246→  // Advanced: Ongoing losses
   247→  const [ongoingLoss, setOngoingLoss] = useState<number>(0);
   248→
   249→  // Advanced: Surgery rate
   250→  const [surgeryRate, setSurgeryRate] = useState<number>(0);
   251→
   252→  // Collapsible states
   253→  const [showDehydration, setShowDehydration] = useState(false);
   254→  const [showOngoingLoss, setShowOngoingLoss] = useState(false);
   255→  const [showSurgery, setShowSurgery] = useState(false);
   256→
   257→  // Results
   258→  const [result, setResult] = useState<FluidResult | null>(null);
   259→  const [error, setError] = useState<string | null>(null);
   260→
   261→  // Live calculation (no API, no localStorage)
   262→  const calculatedResult = useMemo((): FluidResult | null => {
   263→    const w = parseFloat(weight);
   264→    if (!w || w <= 0) return null;
   265→
   266→    let maintenanceDaily: number;
   267→    let formulaUsed: string;
   268→    let maintenanceHourlyMin: number;
   269→    let maintenanceHourlyMax: number;
   270→    let weightRange: string;
   271→    let weightRangeRate: string;
   272→
   273→    if (method === 'standard') {
   274→      const rate = getWeightBasedRate(w);
   275→      maintenanceHourlyMin = rate.hourlyMin;
   276→      maintenanceHourlyMax = rate.hourlyMax;
   277→      // Use midpoint for calculation
   278→      const hourlyRate = (rate.hourlyMin + rate.hourlyMax) / 2;
   279→      maintenanceDaily = hourlyRate * w * 24;
   280→      formulaUsed = `Rango por peso (${rate.range}): ${rate.dailyMin}–${rate.dailyMax} mL/kg/día`;
   281→      weightRange = rate.range;
   282→      weightRangeRate = `${rate.hourlyMin}–${rate.hourlyMax} mL/kg/hr`;
   283→    } else {
   284→      maintenanceDaily = simplifiedFormula(w);
   285→      const hourlyRate = maintenanceDaily / 24;
   286→      maintenanceHourlyMin = Math.round(hourlyRate * 10) / 10;
   287→      maintenanceHourlyMax = Math.round(hourlyRate * 10) / 10;
   288→      formulaUsed = `Fórmula simplificada: (30 × ${w}kg) + 70 = ${maintenanceDaily} mL/día`;
   289→      weightRange = 'N/A';
   290→      weightRangeRate = `${maintenanceHourlyMin.toFixed(1)} mL/kg/hr`;
   291→    }
   292→
   293→    const maintenanceHourly = maintenanceDaily / 24;
   294→
   295→    // Dehydration correction
   296→    const dehydrationVolume = w * (dehydration / 100) * 1000; // mL
   297→    const correctionHourly = dehydration > 0 ? dehydrationVolume / 24 : 0;
   298→
   299→    // Ongoing losses
   300→    const ongoingLossRate = w * ongoingLoss; // mL/hr
   301→    const ongoingHourly = ongoingLossRate;
   302→
   303→    // Surgery additions
   304→    const surgeryRateMl = w * surgeryRate; // mL/hr
   305→    const surgeryHourly = surgeryRateMl;
   306→
   307→    // Totals
   308→    const totalDaily = maintenanceDaily + dehydrationVolume + (ongoingHourly * 24) + (surgeryHourly * 24);
   309→    const totalHourly = totalDaily / 24;
   310→
   311→    // Drops per minute
   312→    const dripSetNum = parseInt(dripSet);
   313→    const dropsPerMinute = (totalHourly * dripSetNum) / 60;
   314→
   315→    // Infusion times for bag sizes
   316→    const infusionTimes: Record<string, number> = {};
   317→    for (const size of BAG_SIZES) {
   318→      infusionTimes[String(size)] = totalHourly > 0 ? size / totalHourly : Infinity;
   319→    }
   320→
   321→    return {
   322→      species,
   323→      weightKg: w,
   324→      method,
   325→      maintenanceDaily: Math.round(maintenanceDaily * 10) / 10,
   326→      maintenanceHourly: Math.round(maintenanceHourly * 100) / 100,
   327→      maintenanceHourlyMin,
   328→      maintenanceHourlyMax,
   329→      dehydrationVolume: Math.round(dehydrationVolume * 10) / 10,
   330→      ongoingLossRate: Math.round(ongoingLossRate * 100) / 100,
   331→      surgeryRate: Math.round(surgeryRateMl * 100) / 100,
   332→      correctionHourly: Math.round(correctionHourly * 100) / 100,
   333→      ongoingHourly: Math.round(ongoingHourly * 100) / 100,
   334→      surgeryHourly: Math.round(surgeryHourly * 100) / 100,
   335→      totalDaily: Math.round(totalDaily * 10) / 10,
   336→      totalHourly: Math.round(totalHourly * 100) / 100,
   337→      dropsPerMinute: Math.round(dropsPerMinute * 100) / 100,
   338→      infusionTimes,
   339→      formulaUsed,
   340→      weightRange,
   341→      weightRangeRate,
   342→    };
   343→  }, [weight, species, method, dripSet, dehydration, ongoingLoss, surgeryRate]);
   344→
   345→  const handleCalculate = () => {
   346→    const w = parseFloat(weight);
   347→    if (!w || w <= 0) {
   348→      setError('Ingrese un peso válido mayor a 0');
   349→      return;
   350→    }
   351→    if (w > 200) {
   352→      setError('El peso máximo permitido es 200 kg');
   353→      return;
   354→    }
   355→    setError(null);
   356→    setResult(calculatedResult);
   357→  };
   358→
   359→  const handleReset = () => {
   360→    setWeight('');
   361→    setDehydration(0);
   362→    setOngoingLoss(0);
   363→    setSurgeryRate(0);
   364→    setResult(null);
   365→    setError(null);
   366→    setShowDehydration(false);
   367→    setShowOngoingLoss(false);
   368→    setShowSurgery(false);
   369→  };
   370→
   371→  const hasAdvancedOptions = dehydration > 0 || ongoingLoss > 0 || surgeryRate > 0;
   372→
   373→  const formatTime = (hours: number): string => {
   374→    if (!isFinite(hours)) return '--:--';
   375→    const h = Math.floor(hours);
   376→    const m = Math.round((hours - h) * 60);
   377→    return `${h}:${m.toString().padStart(2, '0')}`;
   378→  };
   379→
   380→  return (
   381→    <div className="space-y-5">
   382→      {/* ─── Step 1: Species ─── */}
   383→      <div>
   384→        <StepHeading num={1}>Especie</StepHeading>
   385→        <div className="grid grid-cols-2 gap-3">
   386→          {(['perro', 'gato'] as const).map((type) => (
   387→            <button
   388→              key={type}
   389→              onClick={() => { setSpecies(type); setResult(null); }}
   390→              className={`p-4 rounded-xl border-2 transition-all duration-200 flex flex-col items-center gap-2 ${
   391→                species === type
   392→                  ? 'border-primary bg-primary/5 shadow-md shadow-primary/10'
   393→                  : 'border-border hover:border-primary/30'
   394→              }`}
   395→            >
   396→              <span className="text-4xl">{type === 'perro' ? '🐕' : '🐈'}</span>
   397→              <span
   398→                className={`font-semibold ${
   399→                  species === type ? 'text-primary' : 'text-muted-foreground'
   400→                }`}
   401→              >
   402→                {type === 'perro' ? 'Perro' : 'Gato'}
   403→              </span>
   404→            </button>
   405→          ))}
   406→        </div>
   407→      </div>
   408→
   409→      {/* ─── Step 2: Weight ─── */}
   410→      <div>
   411→        <StepHeading num={2}>Peso del Paciente</StepHeading>

   416→            value={weight}

   421→            min="0.1"

   428→              kg

   434→      {/* ─── Step 3: Calculation Method ─── */}

   476→        {/* Info box about the selected method */}
