   124→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   125→  const ref = useRef<HTMLSpanElement>(null);
   126→  const isInView = useInView(ref, { once: true });
   127→  const [count, setCount] = useState(0);
   128→
   129→  useEffect(() => {
   130→    if (!isInView) return;
   131→    let start = 0;
   132→    const end = target;
   133→    const duration = 1500;
   134→    const stepTime = 30;
   135→    const steps = duration / stepTime;
   136→    const increment = end / steps;
   137→
   138→    const timer = setInterval(() => {
   139→      start += increment;
   140→      if (start >= end) {
   141→        setCount(end);
   142→        clearInterval(timer);
   143→      } else {
   144→        setCount(Math.floor(start));
   145→      }
   146→    }, stepTime);
   147→    return () => clearInterval(timer);
   148→  }, [isInView, target]);
   149→
   150→  return <span ref={ref}>{count}{suffix}</span>;
   151→}
