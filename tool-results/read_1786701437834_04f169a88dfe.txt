    83→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    84→  const ref = useRef<HTMLSpanElement>(null);
    85→  const isInView = useInView(ref, { once: true });
    86→  const [count, setCount] = useState(0);
    87→
    88→  useEffect(() => {
    89→    if (!isInView) return;
    90→    let start = 0;
    91→    const end = target;
    92→    const duration = 1500;
    93→    const stepTime = 30;
    94→    const steps = duration / stepTime;
    95→    const increment = end / steps;
    96→
    97→    const timer = setInterval(() => {
    98→      start += increment;
    99→      if (start >= end) {
   100→        setCount(end);
   101→        clearInterval(timer);
   102→      } else {
   103→        setCount(Math.floor(start));
   104→      }
   105→    }, stepTime);
   106→    return () => clearInterval(timer);
   107→  }, [isInView, target]);
   108→
   109→  return <span ref={ref}>{count}{suffix}</span>;
   110→}
