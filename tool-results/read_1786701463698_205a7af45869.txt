    83→    83→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    84→    84→  const ref = useRef<HTMLSpanElement>(null);
    85→    85→  const isInView = useInView(ref, { once: true });
    86→    86→  const [count, setCount] = useState(0);
    87→    87→
    88→    88→  useEffect(() => {
    89→    89→    if (!isInView) return;
    90→    90→    let start = 0;
    91→    91→    const end = target;
    92→    92→    const duration = 1500;
    93→    93→    const stepTime = 30;
    94→    94→    const steps = duration / stepTime;
    95→    95→    const increment = end / steps;
    96→    96→
    97→    97→    const timer = setInterval(() => {
    98→    98→      start += increment;
    99→    99→      if (start >= end) {
   100→   100→        setCount(end);
   101→   101→        clearInterval(timer);
   102→   102→      } else {
   103→   103→        setCount(Math.floor(start));
   104→   104→      }
   105→   105→    }, stepTime);
   106→   106→    return () => clearInterval(timer);
   107→   107→  }, [isInView, target]);
   108→   108→
   109→   109→  return <span ref={ref}>{count}{suffix}</span>;
   110→   110→}
