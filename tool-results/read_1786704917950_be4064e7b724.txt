    85→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    86→  const ref = useRef<HTMLSpanElement>(null);
    87→  const isInView = useInView(ref, { once: true });
    88→  const [count, setCount] = useState(0);
    89→
    90→  useEffect(() => {
    91→    if (!isInView) return;
    92→    let start = 0;
    93→    const end = target;
    94→    const duration = 1500;
    95→    const stepTime = 30;
    96→    const steps = duration / stepTime;
    97→    const increment = end / steps;
    98→
    99→    const timer = setInterval(() => {
   100→      start += increment;
   101→      if (start >= end) {
   102→        setCount(end);
   103→        clearInterval(timer);
   104→      } else {
   105→        setCount(Math.floor(start));
   106→      }
   107→    }, stepTime);
   108→    return () => clearInterval(timer);
   109→  }, [isInView, target]);
   110→
   111→  return <span ref={ref}>{count}{suffix}</span>;
   112→}
