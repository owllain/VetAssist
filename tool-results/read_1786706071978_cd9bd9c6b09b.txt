   101→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   102→  const ref = useRef<HTMLSpanElement>(null);
   103→  const isInView = useInView(ref, { once: true });
   104→  const [count, setCount] = useState(0);
   105→
   106→  useEffect(() => {
   107→    if (!isInView) return;
   108→    let start = 0;
   109→    const end = target;
   110→    const duration = 1500;
   111→    const stepTime = 30;
   112→    const steps = duration / stepTime;
   113→    const increment = end / steps;
   114→
   115→    const timer = setInterval(() => {
   116→      start += increment;
   117→      if (start >= end) {
   118→        setCount(end);
   119→        clearInterval(timer);
   120→      } else {
   121→        setCount(Math.floor(start));
   122→      }
   123→    }, stepTime);
   124→    return () => clearInterval(timer);
   125→  }, [isInView, target]);
   126→
   127→  return <span ref={ref}>{count}{suffix}</span>;
   128→}
