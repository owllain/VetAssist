   132→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   133→  const ref = useRef<HTMLSpanElement>(null);
   134→  const isInView = useInView(ref, { once: true });
   135→  const [count, setCount] = useState(0);
   136→
   137→  useEffect(() => {
   138→    if (!isInView) return;
   139→    let start = 0;
   140→    const end = target;
   141→    const duration = 1500;
   142→    const stepTime = 30;
   143→    const steps = duration / stepTime;
   144→    const increment = end / steps;
   145→
   146→    const timer = setInterval(() => {
   147→      start += increment;
   148→      if (start >= end) {
   149→        setCount(end);
   150→        clearInterval(timer);
   151→      } else {
   152→        setCount(Math.floor(start));
   153→      }
   154→    }, stepTime);
   155→    return () => clearInterval(timer);
   156→  }, [isInView, target]);
   157→
   158→  return <span ref={ref}>{count}{suffix}</span>;
   159→}
