    84→
    85→export default function MedicationCalculator() {
    86→  const [animalType, setAnimalType] = useState<AnimalType>('perro');
    87→  const [weight, setWeight] = useState('');
    88→  const [weightUnit, setWeightUnit] = useState<'kg' | 'lb'>('kg');
    89→  const [selectedCategory, setSelectedCategory] = useState<string | null>(null);
    90→  const [selectedMedication, setSelectedMedication] = useState<Medication | null>(null);
    91→  const [result, setResult] = useState<CalcResult | null>(null);
    92→  const [loading, setLoading] = useState(false);
    93→  const [error, setError] = useState<string | null>(null);
    94→  const [searchQuery, setSearchQuery] = useState('');
    95→  const history = useHistory('medication');
    96→  const addHistory = useAddHistory();
    97→  const [showFavorites, setShowFavorites] = useState(false);
    98→  const [showRefTable, setShowRefTable] = useState(false);
    99→  const [showPatientPanel, setShowPatientPanel] = useState(false);
   100→  const [copied, setCopied] = useState(false);
   101→  const clearHistory = useClearHistory();
   102→  const toggleFav = useToggleFavorite();
   103→  const favorites = useFavorites();
   104→  const favIdSet = new Set(favorites.map(f => f.medicationId));
   105→
   106→  const handleSelectFavorite = useCallback((fav: FavoriteMed) => {
   107→    setSelectedCategory(fav.category);
   108→    setSearchQuery('');
   109→    const med = medications.find((m) => m.id === fav.medicationId);
   110→    if (med) {
   111→      setSelectedMedication(med);
   112→      setResult(null);
   113→    }
   114→  }, []);
   115→
   116→
   117→
   118→  const filteredMedications = useMemo(() => {
   119→    if (!selectedCategory) return [];

   121→    if (searchQuery.trim()) {
   122→      const q = searchQuery.toLowerCase();
   123→      meds = meds.filter(
   124→        (m) =>
   125→          m.name.toLowerCase().includes(q) ||
   126→          m.genericName.toLowerCase().includes(q) ||
   127→          (m.brandNames && m.brandNames.some((b) => b.toLowerCase().includes(q)))
   128→      );
   129→    }
   130→    return meds;
   131→  }, [selectedCategory, searchQuery]);
   132→
   133→  const canCalculate = selectedMedication && weight && parseFloat(weight) > 0;
   134→
   135→  const handleCalculate = async () => {
   136→    if (!selectedMedication || !weight) return;
   137→    setLoading(true);
   138→    setError(null);
   139→    setResult(null);
   140→
   141→    try {
   142→      const res = await fetch('/api/calculate-medication', {
   143→        method: 'POST',
   144→        headers: { 'Content-Type': 'application/json' },
   145→        body: JSON.stringify({
   146→          medicationId: selectedMedication.id,
   147→          animalType,
   148→          weight: parseFloat(weight),
   149→          weightUnit,
   150→        }),
   151→      });
   152→
   153→      const data = await res.json();
   154→
   155→      if (!res.ok) {
   156→        setError(data.error || 'Error al calcular la dosis');
   157→        return;
   158→      }
   159→
   160→      setResult(data);
   161→      // Save to history
   162→      const unit = data.calculatedDose.unit.split('/')[0];
   163→      addHistory({
   164→        type: 'medication',
   165→        timestamp: Date.now(),
   166→        summary: `${data.medication.name} | ${animalType} ${data.weightKg}kg | ${data.calculatedDose.recommended}${unit}`,
   167→      });
   168→      // history auto-updates via useSyncExternalStore
   169→    } catch {
   170→      setError('Error de conexión. Intente de nuevo.');
   171→    } finally {
   172→      setLoading(false);
   173→    }
   174→  };
   175→
   176→  const handleReset = () => {
   177→    setSelectedMedication(null);
   178→    setResult(null);
   179→    setError(null);
   180→  };
   181→
   182→  return (
   183→    <div className="space-y-6">
   184→      {/* Recent History */}
   185→      {history.length > 0 && !result && (
   186→        <div className="no-print">
   187→          <div className="flex items-center justify-between gap-2 mb-2">
   188→            <div className="flex items-center gap-2">
   189→              <Clock size={16} weight="Outline" className="text-muted-foreground" />
   190→              <span className="text-sm font-medium text-muted-foreground">Consultas recientes</span>
   191→            </div>
   192→            <button
   193→              onClick={clearHistory}
   194→              className="flex items-center gap-1 text-xs text-muted-foreground hover:text-destructive transition-colors"
   195→              title="Limpiar historial"
   196→            >
   197→              <Trash size={12} weight="Outline" />
   198→              Limpiar
   199→            </button>
   200→          </div>
   201→          <div className="flex gap-2 overflow-x-auto pb-1">
   202→            {history.slice(0, 5).map((h, i) => (
   203→              <div
   204→                key={i}
   205→                className="flex-shrink-0 bg-muted/60 rounded-lg px-3 py-1.5 text-xs text-muted-foreground border border-border/50"
   206→              >
   207→                {h.summary}
   208→              </div>
   209→            ))}
   210→          </div>
   211→        </div>
   212→      )}
   213→
   214→      {/* Step 1: Animal Type */}
   215→      <div>
   216→        <StepHeading num={1}>Tipo de Animal</StepHeading>
   217→        <div className="grid grid-cols-2 gap-3">
   218→          {(['perro', 'gato'] as const).map((type) => (
   219→            <button
   220→              key={type}
   221→              onClick={() => { setAnimalType(type); setSelectedMedication(null); setResult(null); }}
   222→              className={`p-4 rounded-xl border-2 transition-all duration-200 flex flex-col items-center gap-2 ${
   223→                animalType === type
   224→                  ? 'border-primary bg-primary/5 shadow-md shadow-primary/10'
   225→                  : 'border-border hover:border-primary/30'
   226→              }`}
   227→            >
   228→              {type === 'perro' ? (
   229→                <span className="text-4xl">🐕</span>
   230→              ) : (
   231→                <Cat size={36} weight="Outline" color={animalType === 'gato' ? 'oklch(0.55 0.15 165)' : 'oklch(0.5 0.02 165)'} />
   232→              )}
   233→              <span className={`font-semibold ${animalType === type ? 'text-primary' : 'text-muted-foreground'}`}>
   234→                {type === 'perro' ? 'Perro' : 'Gato'}
   235→              </span>
   236→            </button>
   237→          ))}
   238→        </div>
   239→      </div>
   240→
   241→      {/* Step 2: Weight */}
   242→      <div>
   243→        <StepHeading num={2}>Peso del Animal</StepHeading>
   244→        <div className="flex gap-3 items-center">
   245→          <Input
   246→            type="number"
   247→            placeholder="Ej: 5"
   248→            value={weight}
   249→            onChange={(e) => { setWeight(e.target.value); setResult(null); }}
   250→            min="0.1"
   251→            step="0.1"
   252→            className="flex-1 h-12 text-lg"
   253→          />
   254→          <div className="flex rounded-lg border border-border overflow-hidden">
   255→            {(['kg', 'lb'] as const).map((u) => (
   256→              <button
   257→                key={u}
   258→                onClick={() => setWeightUnit(u)}
   259→                className={`px-4 py-3 text-sm font-semibold transition-colors ${
   260→                  weightUnit === u ? 'bg-primary text-primary-foreground' : 'bg-card hover:bg-muted'
   261→                }`}
   262→              >
   263→                {u}
   264→              </button>
   265→            ))}
   266→          </div>
   267→          <Button
   268→            variant="outline"
   269→            size="icon"
   270→            className={`h-12 w-12 flex-shrink-0 transition-colors ${showPatientPanel ? 'bg-primary/10 border-primary/30' : ''}`}
   271→            onClick={() => setShowPatientPanel(!showPatientPanel)}
   272→            title="Perfiles de pacientes"
   273→          >
   274→            <User size={18} weight={showPatientPanel ? 'Fill' : 'Outline'} className={showPatientPanel ? 'text-primary' : ''} />
   275→          </Button>
   276→        </div>
   277→      </div>
   278→
   279→      {/* Patient Profiles (collapsible) */}
   280→      <Collapsible open={showPatientPanel} onOpenChange={setShowPatientPanel}>
   281→        <CollapsibleContent>
   282→          <motion.div
   283→            initial={{ opacity: 0, height: 0 }}
   284→            animate={{ opacity: 1, height: 'auto' }}
   285→            exit={{ opacity: 0, height: 0 }}
   286→            transition={{ duration: 0.2 }}
   287→          >
   288→            <Card className="glass-card">
   289→              <CardContent className="p-4">
   290→                <PatientProfiles
   291→                  species={animalType}
   292→                  onSelectPatient={(patient: Patient) => {
   293→                    setWeight(String(patient.weight));
   294→                    setWeightUnit(patient.weightUnit);
   295→                    setAnimalType(patient.species);
   296→                    setResult(null);
   297→                  }}
   298→                  currentWeight={weight}
   299→                />
   300→              </CardContent>
   301→            </Card>
   302→          </motion.div>
   303→        </CollapsibleContent>
   304→      </Collapsible>
   305→
   306→      {/* Protocol Quick Templates */}
   307→      {!selectedCategory && !result && (
   308→        <div>
   309→          <ProtocolTemplates
   310→            animalType={animalType}
   311→            onSelectProtocol={(protocol) => {
   312→              if (protocol.drugs.length > 0) {
   313→                const firstDrug = medications.find((m) => m.id === protocol.drugs[0].medicationId);
   314→                if (firstDrug) {
   315→                  setSelectedCategory(firstDrug.categoryId);
   316→                  setSelectedMedication(firstDrug);
   317→                }
   318→              }
   319→            }}
   320→          />
   321→        </div>
   322→      )}
   323→
   324→      {/* Step 3: Category Selection */}
   325→      <div>
   326→        <div className="flex items-center justify-between gap-3 mb-3">
   327→          <StepHeading num={3}>Categoría de Medicamento</StepHeading>
   328→          <Button
   329→            variant="outline"
   330→            size="sm"
   331→            onClick={() => setShowFavorites(true)}
   332→            className="flex-shrink-0 gap-1.5 text-xs no-print"
   333→          >
   334→            <Star size={14} weight="Fill" className="text-amber-500" />
   335→            Favoritos
   336→          </Button>
   337→        </div>
   338→        <div className="flex gap-2 overflow-x-auto pb-2 -mx-1 px-1 no-print">
   339→          {medicationCategories.map((cat) => (
   340→            <button
   341→              key={cat.id}
   342→              onClick={() => {
   343→                setSelectedCategory(cat.id === selectedCategory ? null : cat.id);
   344→                setSelectedMedication(null);
   345→                setResult(null);
   346→                setSearchQuery('');
   347→              }}
   348→              className={`flex-shrink-0 flex items-center gap-2 px-4 py-2.5 rounded-full text-sm font-medium border transition-all duration-200 whitespace-nowrap ${
   349→                selectedCategory === cat.id
   350→                  ? 'bg-primary text-primary-foreground border-primary shadow-md shadow-primary/20'
   351→                  : 'bg-card border-border hover:border-primary/30 hover:bg-primary/5'
   352→              }`}
   353→            >
   354→              <span>{cat.icon}</span>
   355→              <span>{cat.name}</span>
   356→            </button>
   357→          ))}
   358→        </div>
   359→      </div>
   360→
   361→      {/* Step 4: Medication List with Search */}
   362→      <AnimatePresence mode="wait">
   363→        {selectedCategory && filteredMedications.length > 0 && (
   364→          <motion.div
   365→            key={selectedCategory}
   366→            initial={{ opacity: 0, y: 10 }}
   367→            animate={{ opacity: 1, y: 0 }}
   368→            exit={{ opacity: 0, y: -10 }}
   369→            transition={{ duration: 0.2 }}
   370→          >
   371→            <div className="flex items-center justify-between gap-3 mb-3">
   372→              <StepHeading num={4}>Seleccione un Medicamento</StepHeading>
   373→              <div className="relative w-40 flex-shrink-0 no-print">
   374→                <Search size={14} weight="Outline" className="absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground" />
   375→                <Input
   376→                  type="text"
   377→                  placeholder="Buscar..."
   378→                  value={searchQuery}
   379→                  onChange={(e) => setSearchQuery(e.target.value)}
   380→                  className="h-8 text-sm pl-8"
   381→                />
   382→              </div>
   383→            </div>
   384→
   385→            {/* Dose Reference Table - collapsible */}
   386→            <Collapsible open={showRefTable} onOpenChange={setShowRefTable} className="mb-3 no-print">
   387→              <CollapsibleTrigger asChild>
   388→                <Button
   389→                  variant="ghost"
   390→                  size="sm"
   391→                  className="w-full justify-between gap-2 text-xs text-muted-foreground hover:text-foreground h-8 px-2"
   392→                >
   393→                  <span className="flex items-center gap-1.5">
   394→                    <ClipboardList size={14} weight="Outline" />
   395→                    Ver tabla de referencia
   396→                  </span>
   397→                  <motion.span
   398→                    animate={{ rotate: showRefTable ? 180 : 0 }}
   399→                    transition={{ duration: 0.2 }}
   400→                  >
   401→                    <ChevronDown size={14} weight="Outline" />
   402→                  </motion.span>
   403→                </Button>
   404→              </CollapsibleTrigger>
   405→              <CollapsibleContent>
   406→                <motion.div
   407→                  initial={{ opacity: 0, height: 0 }}
   408→                  animate={{ opacity: 1, height: 'auto' }}
   409→                  exit={{ opacity: 0, height: 0 }}
   410→                  transition={{ duration: 0.2 }}
   411→                >
   412→                  <DoseReferenceTable categoryId={selectedCategory} animalType={animalType} />
   413→                </motion.div>
   414→              </CollapsibleContent>
   415→            </Collapsible>
   416→
   417→            <div className="grid gap-2.5 max-h-[400px] overflow-y-auto pr-1">
   418→              {filteredMedications.map((med) => {
   419→                const isAvailable = med.species.includes(animalType);
   420→                const isSelected = selectedMedication?.id === med.id;
   421→                const medIsFav = favIdSet.has(med.id);
   422→
   423→                return (
   424→                  <Card
   425→                    key={med.id}
   426→                    className={`med-card cursor-pointer ${isSelected ? 'selected' : ''} ${
   427→                      !isAvailable ? 'opacity-40 pointer-events-none' : ''
   428→                    }`}
   429→                    onClick={() => {
   430→                      if (!isAvailable) return;
   431→                      setSelectedMedication(isSelected ? null : med);
   432→                      setResult(null);
   433→                    }}
   434→                  >
   435→                    <CardContent className="p-3.5">
   436→                      <div className="flex items-start justify-between gap-2">
   437→                        <div className="flex-1 min-w-0">
   438→                          <div className="flex items-center gap-2 flex-wrap">
   439→                            <h4 className="font-bold text-sm">{med.name}</h4>
   440→                            {med.species.map((s) => (
   441→                              <Badge
   442→                                key={s}
   443→                                variant={s === animalType ? 'default' : 'secondary'}
   444→                                className="text-[10px] px-1.5 py-0"
   445→                              >
   446→                                {s === 'perro' ? '🐕' : '🐈'} {s}
   447→                              </Badge>
   448→                            ))}
   449→                          </div>
   450→                          <p className="text-xs text-muted-foreground mt-0.5">{med.genericName}</p>
   451→                          <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   452→                            <Badge variant="outline" className="text-[10px] font-semibold text-primary border-primary/30">
   453→                              {med.doseMin === med.doseMax
   454→                                ? `${med.doseMin} ${med.unit}`
   455→                                : `${med.doseMin}–${med.doseMax} ${med.unit}`}
   456→                            </Badge>
   457→                            {med.route.map((r) => (
   458→                              <Badge key={r} variant="secondary" className="text-[10px] px-1.5 py-0">
   459→                                {r}
   460→                              </Badge>
   461→                            ))}
   462→                          </div>
   463→                        </div>
   464→                        <div className="flex items-center gap-1 flex-shrink-0 mt-1">
   465→                          <button
   466→                            onClick={(e) => {
   467→                              e.stopPropagation();
   468→                              toggleFav(med);
   469→                            }}
   470→                            className={`p-1 rounded-md transition-all hover:bg-amber-100 dark:hover:bg-amber-900/30 ${
   471→                              medIsFav ? 'text-amber-500' : 'text-muted-foreground/30 hover:text-amber-400'
   472→                            }`}
   473→                            title={medIsFav ? 'Quitar de favoritos' : 'Agregar a favoritos'}
   474→                          >
   475→                            <Star size={16} weight={medIsFav ? 'fill' : 'outline'} />
   476→                          </button>
   477→                          <ChevronRight
   478→                            size={18} weight="Outline"
   479→                            className={`transition-all ${isSelected ? 'text-primary rotate-90' : 'text-muted-foreground/40'}`}
   480→                          />
   481→                        </div>
   482→                      </div>
   483→                    </CardContent>
   484→                  </Card>
   485→                );
   486→              })}
   487→            </div>
   488→          </motion.div>
   489→        )}
   490→        {selectedCategory && filteredMedications.length === 0 && (
   491→          <motion.div
   492→            key="empty"
   493→            initial={{ opacity: 0 }} animate={{ opacity: 1 }}
   494→            className="text-center py-8 text-muted-foreground"
   495→          >
   496→            <Search size={32} weight="Outline" className="mx-auto mb-2 opacity-30" />
   497→            <p className="text-sm">No se encontraron medicamentos</p>
   498→          </motion.div>
   499→        )}
   500→      </AnimatePresence>
   501→
   502→      {/* Calculate Button */}
   503→      <AnimatePresence>
   504→        {selectedMedication && (
   505→          <motion.div
   506→            initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}
   507→            className="flex flex-col items-center gap-3 no-print"
   508→          >
   509→            <Button
   510→              size="lg"
   511→              onClick={handleCalculate}
   512→              disabled={!canCalculate || loading}
   513→              className="vet-pulse text-lg px-8 py-6 bg-primary hover:bg-primary/90 text-primary-foreground rounded-xl shadow-lg shadow-primary/20"
   514→            >
   515→              {loading ? (
   516→                <span className="flex items-center gap-2">
   517→                  <span className="animate-spin">⚙️</span>
   518→                  Calculando...
   519→                </span>
   520→              ) : (
   521→                <span className="flex items-center gap-2">
   522→                  <Calculator size={22} weight="Outline" />
   523→                  CALCULAR DOSIS
   524→                </span>
   525→              )}
   526→            </Button>
   527→            <p className="text-sm text-muted-foreground">
   528→              {selectedMedication.name} — {selectedMedication.genericName}
   529→            </p>
   530→          </motion.div>
   531→        )}
   532→      </AnimatePresence>
   533→
   534→      {/* Drug Interaction Checker */}
   535→      {selectedMedication && !result && (
   536→        <DrugInteractionChecker selectedMedicationId={selectedMedication.id} />
   537→      )}
   538→
   539→      {/* Error */}
   540→      <AnimatePresence>
   541→        {error && (
   542→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }}>
   543→            <Alert variant="destructive">
   544→              <AlertTriangle size={18} weight="Outline" />

   546→            </Alert>
   547→          </motion.div>
   548→        )}
   549→      </AnimatePresence>
   550→
   551→      {/* Result */}
   552→      <AnimatePresence>
   553→        {result && (
   554→          <motion.div
   555→            initial={{ opacity: 0, y: 20, scale: 0.95 }}

   557→            exit={{ opacity: 0, y: -10, scale: 0.95 }}
   558→            transition={{ duration: 0.3 }}
   559→          >
   560→            <Card className="result-card shadow-lg card-shine">
   561→              <CardHeader className="pb-3">
