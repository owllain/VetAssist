     1→'use client';
     2→
     3→import { useState, useMemo, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  HeartPulse,
     7→  Pill,
     8→  Stethoscope,
     9→  Shield,
    10→  Syringe,
    11→  Cat,
    12→  Scale,
    13→  Calculator,
    14→  AlertTriangle,
    15→  CircleInfo,
    16→  MedicalKit,
    17→  Paw,
    18→  ChevronRight,
    19→  ChevronDown,
    20→  Search,
    21→  Printer,
    22→  Clock,
    23→  Star,
    24→  ClipboardList,
    25→} from 'reicon-react';
    26→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    27→import { Button } from '@/components/ui/button';
    28→import { Input } from '@/components/ui/input';
    29→import { Badge } from '@/components/ui/badge';
    30→import { Alert, AlertDescription } from '@/components/ui/alert';
    31→import { Collapsible, CollapsibleTrigger, CollapsibleContent } from '@/components/ui/collapsible';
    32→import {
    33→  type Medication,
    34→  type AnimalType,
    35→  medicationCategories,
    36→  getMedicationsByCategory,
    37→  medications,
    38→} from '@/lib/medications';
    39→import { addFavorite, removeFavorite, isFavorite } from '@/lib/favorites';
    40→import type { FavoriteMed } from '@/lib/favorites';
    41→import FavoritesPanel from './FavoritesPanel';
    42→import DoseReferenceTable from './DoseReferenceTable';
    43→import { useHistory, useAddHistory } from '@/lib/use-history-store';
    44→
    45→interface CalcResult {
    46→  medication: {
    47→    id: string;
    48→    name: string;
    49→    genericName: string;
    50→    category: string;
    51→    brandNames: string[];
    52→  };
    53→  calculatedDose: {
    54→    min: number;
    55→    max: number;
    56→    recommended: number;
    57→    unit: string;
    58→  };
    59→  weightKg: number;
    60→  routes: string[];
    61→  frequency: string[];
    62→  notes: string;
    63→}
    64→
    65→
    66→
    67→function StepHeading({ num, children }: { num: number; children: React.ReactNode }) {
    68→  return (
    69→    <h3 className="text-base font-semibold mb-3 flex items-center gap-2.5">
    70→      <span className="step-number">{num}</span>
    71→      {children}
    72→    </h3>
    73→  );
    74→}
    75→
    76→export default function MedicationCalculator() {
    77→  const [animalType, setAnimalType] = useState<AnimalType>('perro');
    78→  const [weight, setWeight] = useState('');
    79→  const [weightUnit, setWeightUnit] = useState<'kg' | 'lb'>('kg');
    80→  const [selectedCategory, setSelectedCategory] = useState<string | null>(null);
    81→  const [selectedMedication, setSelectedMedication] = useState<Medication | null>(null);
    82→  const [result, setResult] = useState<CalcResult | null>(null);
    83→  const [loading, setLoading] = useState(false);
    84→  const [error, setError] = useState<string | null>(null);
    85→  const [searchQuery, setSearchQuery] = useState('');
    86→  const history = useHistory('medication');
    87→  const addHistory = useAddHistory();
    88→  const [showFavorites, setShowFavorites] = useState(false);
    89→  const [showRefTable, setShowRefTable] = useState(false);
    90→  const [favTick, setFavTick] = useState(0);
    91→
    92→  const handleSelectFavorite = useCallback((fav: FavoriteMed) => {
    93→    setSelectedCategory(fav.category);
    94→    setSearchQuery('');
    95→    const med = medications.find((m) => m.id === fav.medicationId);
    96→    if (med) {
    97→      setSelectedMedication(med);
    98→      setResult(null);
    99→    }
   100→  }, []);
   101→
   102→  const toggleFav = useCallback((med: Medication) => {
   103→    if (isFavorite(med.id)) {
   104→      removeFavorite(med.id);
   105→    } else {
   106→      addFavorite({
   107→        medicationId: med.id,
   108→        name: med.name,
   109→        genericName: med.genericName,
   110→        category: med.categoryId,
   111→        species: med.species,
   112→      });
   113→    }
   114→    setFavTick((t) => t + 1);
   115→  }, []);
   116→
   117→  const filteredMedications = useMemo(() => {
   118→    if (!selectedCategory) return [];
   119→    let meds = getMedicationsByCategory(selectedCategory);
   120→    if (searchQuery.trim()) {
   121→      const q = searchQuery.toLowerCase();
   122→      meds = meds.filter(
   123→        (m) =>
   124→          m.name.toLowerCase().includes(q) ||
   125→          m.genericName.toLowerCase().includes(q) ||
   126→          (m.brandNames && m.brandNames.some((b) => b.toLowerCase().includes(q)))
   127→      );
   128→    }
   129→    return meds;
   130→  }, [selectedCategory, searchQuery, favTick]);
   131→
   132→  const canCalculate = selectedMedication && weight && parseFloat(weight) > 0;
   133→
   134→  const handleCalculate = async () => {
   135→    if (!selectedMedication || !weight) return;
   136→    setLoading(true);
   137→    setError(null);
   138→    setResult(null);
   139→
   140→    try {
   141→      const res = await fetch('/api/calculate-medication', {
   142→        method: 'POST',
   143→        headers: { 'Content-Type': 'application/json' },
   144→        body: JSON.stringify({
   145→          medicationId: selectedMedication.id,
   146→          animalType,
   147→          weight: parseFloat(weight),
   148→          weightUnit,
   149→        }),
   150→      });
   151→
   152→      const data = await res.json();
   153→
   154→      if (!res.ok) {
   155→        setError(data.error || 'Error al calcular la dosis');
   156→        return;
   157→      }
   158→
   159→      setResult(data);
   160→      // Save to history
   161→      const unit = data.calculatedDose.unit.split('/')[0];
   162→      addHistory({
   163→        type: 'medication',
   164→        timestamp: Date.now(),
   165→        summary: `${data.medication.name} | ${animalType} ${data.weightKg}kg | ${data.calculatedDose.recommended}${unit}`,
   166→      });
   167→      // history auto-updates via useSyncExternalStore
   168→    } catch {
   169→      setError('Error de conexión. Intente de nuevo.');
   170→    } finally {
   171→      setLoading(false);
   172→    }
   173→  };
   174→
   175→  const handleReset = () => {
   176→    setSelectedMedication(null);
   177→    setResult(null);
   178→    setError(null);
   179→  };
   180→
   181→  return (
   182→    <div className="space-y-6">
   183→      {/* Recent History */}
   184→      {history.length > 0 && !result && (
   185→        <div className="no-print">
   186→          <div className="flex items-center gap-2 mb-2">
   187→            <Clock size={16} weight="outline" className="text-muted-foreground" />
   188→            <span className="text-sm font-medium text-muted-foreground">Consultas recientes</span>
   189→          </div>
   190→          <div className="flex gap-2 overflow-x-auto pb-1">
   191→            {history.slice(0, 5).map((h, i) => (
   192→              <div
   193→                key={i}
   194→                className="flex-shrink-0 bg-muted/60 rounded-lg px-3 py-1.5 text-xs text-muted-foreground border border-border/50"
   195→              >
   196→                {h.summary}
   197→              </div>
   198→            ))}
   199→          </div>
   200→        </div>
   201→      )}
   202→
   203→      {/* Step 1: Animal Type */}
   204→      <div>
   205→        <StepHeading num={1}>Tipo de Animal</StepHeading>
   206→        <div className="grid grid-cols-2 gap-3">
   207→          {(['perro', 'gato'] as const).map((type) => (
   208→            <button
   209→              key={type}
   210→              onClick={() => { setAnimalType(type); setSelectedMedication(null); setResult(null); }}
   211→              className={`p-4 rounded-xl border-2 transition-all duration-200 flex flex-col items-center gap-2 ${
   212→                animalType === type
   213→                  ? 'border-primary bg-primary/5 shadow-md shadow-primary/10'
   214→                  : 'border-border hover:border-primary/30'
   215→              }`}
   216→            >
   217→              {type === 'perro' ? (
   218→                <span className="text-4xl">🐕</span>
   219→              ) : (
   220→                <Cat size={36} weight="outline" color={animalType === 'gato' ? 'oklch(0.55 0.15 165)' : 'oklch(0.5 0.02 165)'} />
   221→              )}
   222→              <span className={`font-semibold ${animalType === type ? 'text-primary' : 'text-muted-foreground'}`}>
   223→                {type === 'perro' ? 'Perro' : 'Gato'}
   224→              </span>
   225→            </button>
   226→          ))}
   227→        </div>
   228→      </div>
   229→
   230→      {/* Step 2: Weight */}
   231→      <div>
   232→        <StepHeading num={2}>Peso del Animal</StepHeading>
   233→        <div className="flex gap-3 items-center">
   234→          <Input
   235→            type="number"
   236→            placeholder="Ej: 5"
   237→            value={weight}
   238→            onChange={(e) => { setWeight(e.target.value); setResult(null); }}
   239→            min="0.1"
   240→            step="0.1"
   241→            className="flex-1 h-12 text-lg"
   242→          />
   243→          <div className="flex rounded-lg border border-border overflow-hidden">
   244→            {(['kg', 'lb'] as const).map((u) => (
   245→              <button
   246→                key={u}
   247→                onClick={() => setWeightUnit(u)}
   248→                className={`px-4 py-3 text-sm font-semibold transition-colors ${
   249→                  weightUnit === u ? 'bg-primary text-primary-foreground' : 'bg-card hover:bg-muted'
   250→                }`}
   251→              >
   252→                {u}
   253→              </button>
   254→            ))}
   255→          </div>
   256→        </div>
   257→      </div>
   258→
   259→      {/* Step 3: Category Selection */}
   260→      <div>
   261→        <div className="flex items-center justify-between gap-3 mb-3">
   262→          <StepHeading num={3}>Categoría de Medicamento</StepHeading>
   263→          <Button
   264→            variant="outline"
   265→            size="sm"
   266→            onClick={() => setShowFavorites(true)}
   267→            className="flex-shrink-0 gap-1.5 text-xs no-print"
   268→          >
   269→            <Star size={14} weight="fill" className="text-amber-500" />
   270→            Favoritos
   271→          </Button>
   272→        </div>
   273→        <div className="flex gap-2 overflow-x-auto pb-2 -mx-1 px-1 no-print">
   274→          {medicationCategories.map((cat) => (
   275→            <button
   276→              key={cat.id}
   277→              onClick={() => {
   278→                setSelectedCategory(cat.id === selectedCategory ? null : cat.id);
   279→                setSelectedMedication(null);
   280→                setResult(null);
   281→                setSearchQuery('');
   282→              }}
   283→              className={`flex-shrink-0 flex items-center gap-2 px-4 py-2.5 rounded-full text-sm font-medium border transition-all duration-200 whitespace-nowrap ${
   284→                selectedCategory === cat.id
   285→                  ? 'bg-primary text-primary-foreground border-primary shadow-md shadow-primary/20'
   286→                  : 'bg-card border-border hover:border-primary/30 hover:bg-primary/5'
   287→              }`}
   288→            >
   289→              <span>{cat.icon}</span>
   290→              <span>{cat.name}</span>
   291→            </button>
   292→          ))}
   293→        </div>
   294→      </div>
   295→
   296→      {/* Step 4: Medication List with Search */}
   297→      <AnimatePresence mode="wait">
   298→        {selectedCategory && filteredMedications.length > 0 && (
   299→          <motion.div
   300→            key={selectedCategory}
   301→            initial={{ opacity: 0, y: 10 }}
   302→            animate={{ opacity: 1, y: 0 }}
   303→            exit={{ opacity: 0, y: -10 }}
   304→            transition={{ duration: 0.2 }}
   305→          >
   306→            <div className="flex items-center justify-between gap-3 mb-3">
   307→              <StepHeading num={4}>Seleccione un Medicamento</StepHeading>
   308→              <div className="relative w-40 flex-shrink-0 no-print">
   309→                <Search size={14} weight="outline" className="absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground" />
   310→                <Input
   311→                  type="text"
   312→                  placeholder="Buscar..."
   313→                  value={searchQuery}
   314→                  onChange={(e) => setSearchQuery(e.target.value)}
   315→                  className="h-8 text-sm pl-8"
   316→                />
   317→              </div>
   318→            </div>
   319→
   320→            {/* Dose Reference Table - collapsible */}
   321→            <Collapsible open={showRefTable} onOpenChange={setShowRefTable} className="mb-3 no-print">
   322→              <CollapsibleTrigger asChild>
   323→                <Button
   324→                  variant="ghost"
   325→                  size="sm"
   326→                  className="w-full justify-between gap-2 text-xs text-muted-foreground hover:text-foreground h-8 px-2"
   327→                >
   328→                  <span className="flex items-center gap-1.5">
   329→                    <ClipboardList size={14} weight="outline" />
   330→                    Ver tabla de referencia
   331→                  </span>
   332→                  <motion.span
   333→                    animate={{ rotate: showRefTable ? 180 : 0 }}
   334→                    transition={{ duration: 0.2 }}
   335→                  >
   336→                    <ChevronDown size={14} weight="outline" />
   337→                  </motion.span>
   338→                </Button>
   339→              </CollapsibleTrigger>
   340→              <CollapsibleContent>
   341→                <motion.div
   342→                  initial={{ opacity: 0, height: 0 }}
   343→                  animate={{ opacity: 1, height: 'auto' }}
   344→                  exit={{ opacity: 0, height: 0 }}
   345→                  transition={{ duration: 0.2 }}
   346→                >
   347→                  <DoseReferenceTable categoryId={selectedCategory} animalType={animalType} />
   348→                </motion.div>
   349→              </CollapsibleContent>
   350→            </Collapsible>
   351→
   352→            <div className="grid gap-2.5 max-h-[400px] overflow-y-auto pr-1">
   353→              {filteredMedications.map((med) => {
   354→                const isAvailable = med.species.includes(animalType);
   355→                const isSelected = selectedMedication?.id === med.id;
   356→                const medIsFav = isFavorite(med.id);
   357→
   358→                return (
   359→                  <Card
   360→                    key={med.id}
   361→                    className={`med-card cursor-pointer ${isSelected ? 'selected' : ''} ${
   362→                      !isAvailable ? 'opacity-40 pointer-events-none' : ''
   363→                    }`}
   364→                    onClick={() => {
   365→                      if (!isAvailable) return;
   366→                      setSelectedMedication(isSelected ? null : med);
   367→                      setResult(null);
   368→                    }}
   369→                  >
   370→                    <CardContent className="p-3.5">
   371→                      <div className="flex items-start justify-between gap-2">
   372→                        <div className="flex-1 min-w-0">
   373→                          <div className="flex items-center gap-2 flex-wrap">
   374→                            <h4 className="font-bold text-sm">{med.name}</h4>
   375→                            {med.species.map((s) => (
   376→                              <Badge
   377→                                key={s}
   378→                                variant={s === animalType ? 'default' : 'secondary'}
   379→                                className="text-[10px] px-1.5 py-0"
   380→                              >
   381→                                {s === 'perro' ? '🐕' : '🐈'} {s}
   382→                              </Badge>
   383→                            ))}
   384→                          </div>
   385→                          <p className="text-xs text-muted-foreground mt-0.5">{med.genericName}</p>
   386→                          <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   387→                            <Badge variant="outline" className="text-[10px] font-semibold text-primary border-primary/30">
   388→                              {med.doseMin === med.doseMax
   389→                                ? `${med.doseMin} ${med.unit}`
   390→                                : `${med.doseMin}–${med.doseMax} ${med.unit}`}
   391→                            </Badge>
   392→                            {med.route.map((r) => (
   393→                              <Badge key={r} variant="secondary" className="text-[10px] px-1.5 py-0">
   394→                                {r}
   395→                              </Badge>
   396→                            ))}
   397→                          </div>
   398→                        </div>
   399→                        <div className="flex items-center gap-1 flex-shrink-0 mt-1">
   400→                          <button
   401→                            onClick={(e) => {
   402→                              e.stopPropagation();
   403→                              toggleFav(med);
   404→                            }}
   405→                            className={`p-1 rounded-md transition-all hover:bg-amber-100 dark:hover:bg-amber-900/30 ${
   406→                              medIsFav ? 'text-amber-500' : 'text-muted-foreground/30 hover:text-amber-400'
   407→                            }`}
   408→                            title={medIsFav ? 'Quitar de favoritos' : 'Agregar a favoritos'}
   409→                          >
   410→                            <Star size={16} weight={medIsFav ? 'fill' : 'outline'} />
   411→                          </button>
   412→                          <ChevronRight
   413→                            size={18} weight="outline"
   414→                            className={`transition-all ${isSelected ? 'text-primary rotate-90' : 'text-muted-foreground/40'}`}
   415→                          />
   416→                        </div>
   417→                      </div>
   418→                    </CardContent>
   419→                  </Card>
   420→                );
   421→              })}
   422→            </div>
   423→          </motion.div>
   424→        )}
   425→        {selectedCategory && filteredMedications.length === 0 && (
   426→          <motion.div
   427→            key="empty"
   428→            initial={{ opacity: 0 }} animate={{ opacity: 1 }}
   429→            className="text-center py-8 text-muted-foreground"
   430→          >
   431→            <Search size={32} weight="outline" className="mx-auto mb-2 opacity-30" />
   432→            <p className="text-sm">No se encontraron medicamentos</p>
   433→          </motion.div>
   434→        )}
   435→      </AnimatePresence>
   436→
   437→      {/* Calculate Button */}
   438→      <AnimatePresence>
   439→        {selectedMedication && (
   440→          <motion.div
   441→            initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }}
   442→            className="flex flex-col items-center gap-3 no-print"
   443→          >
   444→            <Button
   445→              size="lg"
   446→              onClick={handleCalculate}
   447→              disabled={!canCalculate || loading}
   448→              className="vet-pulse text-lg px-8 py-6 bg-primary hover:bg-primary/90 text-primary-foreground rounded-xl shadow-lg shadow-primary/20"
   449→            >
   450→              {loading ? (
   451→                <span className="flex items-center gap-2">
   452→                  <span className="animate-spin">⚙️</span>
   453→                  Calculando...
   454→                </span>
   455→              ) : (
   456→                <span className="flex items-center gap-2">
   457→                  <Calculator size={22} weight="outline" />
   458→                  CALCULAR DOSIS
   459→                </span>
   460→              )}
   461→            </Button>
   462→            <p className="text-sm text-muted-foreground">
   463→              {selectedMedication.name} — {selectedMedication.genericName}
   464→            </p>
   465→          </motion.div>
   466→        )}
   467→      </AnimatePresence>
   468→
   469→      {/* Error */}
   470→      <AnimatePresence>
   471→        {error && (
   472→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }}>
   473→            <Alert variant="destructive">
   474→              <AlertTriangle size={18} weight="outline" />
   475→              <AlertDescription>{error}</AlertDescription>
   476→            </Alert>
   477→          </motion.div>
   478→        )}
   479→      </AnimatePresence>
   480→
   481→      {/* Result */}
   482→      <AnimatePresence>
   483→        {result && (
   484→          <motion.div
   485→            initial={{ opacity: 0, y: 20, scale: 0.95 }}
   486→            animate={{ opacity: 1, y: 0, scale: 1 }}
   487→            exit={{ opacity: 0, y: -10, scale: 0.95 }}
   488→            transition={{ duration: 0.3 }}
   489→          >
   490→            <Card className="result-card shadow-lg">
   491→              <CardHeader className="pb-3">
   492→                <div className="flex items-center justify-between">
   493→                  <CardTitle className="flex items-center gap-2 text-primary">
   494→                    <div className="w-8 h-8 rounded-lg bg-primary/10 flex items-center justify-center">
   495→                      <Stethoscope size={18} weight="outline" />
   496→                    </div>
   497→                    Resultado del Cálculo
   498→                  </CardTitle>
   499→                  <div className="flex items-center gap-1">
   500→                    {result.medication.id && (
   501→                      <Button
   502→                        variant="ghost"
   503→                        size="icon"
   504→                        onClick={() => {
   505→                          const med = medications.find((m) => m.id === result.medication.id);
   506→                          if (med) toggleFav(med);
   507→                        }}
   508→                        className={`no-print h-8 w-8 ${
   509→                          isFavorite(result.medication.id)
   510→                            ? 'text-amber-500'
   511→                            : 'text-muted-foreground/40 hover:text-amber-400'
   512→                        }`}
   513→                        title={
   514→                          isFavorite(result.medication.id)
   515→                            ? 'Quitar de favoritos'
   516→                            : 'Agregar a favoritos'
   517→                        }
   518→                      >
   519→                        <Star
   520→                          size={16}
   521→                          weight={isFavorite(result.medication.id) ? 'fill' : 'outline'}
   522→                        />
   523→                      </Button>
   524→                    )}
   525→                    <Button
   526→                      variant="ghost"
   527→                      size="icon"
   528→                      onClick={() => window.print()}
   529→                      className="no-print h-8 w-8 text-muted-foreground hover:text-primary"
   530→                      title="Imprimir"
   531→                    >
   532→                      <Printer size={16} weight="outline" />
   533→                    </Button>
   534→                  </div>
   535→                </div>
   536→              </CardHeader>
   537→              <CardContent className="space-y-4">
   538→                {/* Medication Info */}
   539→                <div className="dose-highlight p-4">
   540→                  <div className="flex items-center gap-3">
   541→                    <div className="w-12 h-12 rounded-xl bg-primary/10 flex items-center justify-center flex-shrink-0">
   542→                      <Pill size={24} color="oklch(0.55 0.15 165)" weight="outline" />
   543→                    </div>
   544→                    <div className="flex-1 min-w-0">
   545→                      <h4 className="font-bold text-lg leading-tight">{result.medication.name}</h4>
   546→                      <p className="text-sm text-muted-foreground">{result.medication.genericName}</p>
   547→                    </div>
   548→                  </div>
   549→                </div>
   550→
   551→                {/* Dose Results */}
   552→                <div className="grid grid-cols-3 gap-3">
   553→                  <div className="bg-primary/5 rounded-xl p-3 text-center border border-primary/10">
   554→                    <p className="text-[10px] text-muted-foreground font-semibold uppercase tracking-wider">Mínima</p>
   555→                    <p className="text-2xl font-extrabold text-primary mt-1">
   556→                      {result.calculatedDose.min}
