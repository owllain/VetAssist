    76→    76→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    77→    77→  const ref = useRef<HTMLSpanElement>(null);
    78→    78→  const isInView = useInView(ref, { once: true });
    79→    79→  const [count, setCount] = useState(0);
    80→    80→
    81→    81→  useEffect(() => {
    82→    82→    if (!isInView) return;
    83→    83→    let start = 0;
    84→    84→    const end = target;
    85→    85→    const duration = 1500;
    86→    86→    const stepTime = 30;
    87→    87→    const steps = duration / stepTime;
    88→    88→    const increment = end / steps;
    89→    89→
    90→    90→    const timer = setInterval(() => {
    91→    91→      start += increment;
    92→    92→      if (start >= end) {
    93→    93→        setCount(end);
    94→    94→        clearInterval(timer);
    95→    95→      } else {
    96→    96→        setCount(Math.floor(start));
    97→    97→      }
    98→    98→    }, stepTime);
    99→    99→    return () => clearInterval(timer);
   100→   100→  }, [isInView, target]);
   101→   101→
   102→   102→  return <span ref={ref}>{count}{suffix}</span>;
   103→   103→}
