    79→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    80→  const ref = useRef<HTMLSpanElement>(null);
    81→  const isInView = useInView(ref, { once: true });
    82→  const [count, setCount] = useState(0);
    83→
    84→  useEffect(() => {
    85→    if (!isInView) return;
    86→    let start = 0;
    87→    const end = target;
    88→    const duration = 1500;
    89→    const stepTime = 30;
    90→    const steps = duration / stepTime;
    91→    const increment = end / steps;
    92→
    93→    const timer = setInterval(() => {
    94→      start += increment;
    95→      if (start >= end) {
    96→        setCount(end);
    97→        clearInterval(timer);
    98→      } else {
    99→        setCount(Math.floor(start));
   100→      }
   101→    }, stepTime);
   102→    return () => clearInterval(timer);
   103→  }, [isInView, target]);
   104→
   105→  return <span ref={ref}>{count}{suffix}</span>;
   106→}
