    82→    82→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    83→    83→  const ref = useRef<HTMLSpanElement>(null);
    84→    84→  const isInView = useInView(ref, { once: true });
    85→    85→  const [count, setCount] = useState(0);
    86→    86→
    87→    87→  useEffect(() => {
    88→    88→    if (!isInView) return;
    89→    89→    let start = 0;
    90→    90→    const end = target;
    91→    91→    const duration = 1500;
    92→    92→    const stepTime = 30;
    93→    93→    const steps = duration / stepTime;
    94→    94→    const increment = end / steps;
    95→    95→
    96→    96→    const timer = setInterval(() => {
    97→    97→      start += increment;
    98→    98→      if (start >= end) {
    99→    99→        setCount(end);
   100→   100→        clearInterval(timer);
   101→   101→      } else {
   102→   102→        setCount(Math.floor(start));
   103→   103→      }
   104→   104→    }, stepTime);
   105→   105→    return () => clearInterval(timer);
   106→   106→  }, [isInView, target]);
   107→   107→
   108→   108→  return <span ref={ref}>{count}{suffix}</span>;
   109→   109→}
