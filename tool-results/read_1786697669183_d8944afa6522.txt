    79→    79→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
    80→    80→  const ref = useRef<HTMLSpanElement>(null);
    81→    81→  const isInView = useInView(ref, { once: true });
    82→    82→  const [count, setCount] = useState(0);
    83→    83→
    84→    84→  useEffect(() => {
    85→    85→    if (!isInView) return;
    86→    86→    let start = 0;
    87→    87→    const end = target;
    88→    88→    const duration = 1500;
    89→    89→    const stepTime = 30;
    90→    90→    const steps = duration / stepTime;
    91→    91→    const increment = end / steps;
    92→    92→
    93→    93→    const timer = setInterval(() => {
    94→    94→      start += increment;
    95→    95→      if (start >= end) {
    96→    96→        setCount(end);
    97→    97→        clearInterval(timer);
    98→    98→      } else {
    99→    99→        setCount(Math.floor(start));
   100→   100→      }
   101→   101→    }, stepTime);
   102→   102→    return () => clearInterval(timer);
   103→   103→  }, [isInView, target]);
   104→   104→
   105→   105→  return <span ref={ref}>{count}{suffix}</span>;
   106→   106→}
