    75→    75→
    76→    76→export default function MedicationCalculator() {
    77→    77→  const [animalType, setAnimalType] = useState<AnimalType>('perro');
    78→    78→  const [weight, setWeight] = useState('');
    79→    79→  const [weightUnit, setWeightUnit] = useState<'kg' | 'lb'>('kg');
    80→    80→  const [selectedCategory, setSelectedCategory] = useState<string | null>(null);
    81→    81→  const [selectedMedication, setSelectedMedication] = useState<Medication | null>(null);
    82→    82→  const [result, setResult] = useState<CalcResult | null>(null);
    83→    83→  const [loading, setLoading] = useState(false);
    84→    84→  const [error, setError] = useState<string | null>(null);
    85→    85→  const [searchQuery, setSearchQuery] = useState('');
    86→    86→  const history = useHistory('medication');
    87→    87→  const addHistory = useAddHistory();
    88→    88→  const [showFavorites, setShowFavorites] = useState(false);
    89→    89→  const [showRefTable, setShowRefTable] = useState(false);
    90→    90→  const [favTick, setFavTick] = useState(0);
    91→    91→
    92→    92→  const handleSelectFavorite = useCallback((fav: FavoriteMed) => {
    93→    93→    setSelectedCategory(fav.category);
    94→    94→    setSearchQuery('');
    95→    95→    const med = medications.find((m) => m.id === fav.medicationId);
    96→    96→    if (med) {
    97→    97→      setSelectedMedication(med);
    98→    98→      setResult(null);
    99→    99→    }
   100→   100→  }, []);
   101→   101→
   102→   102→  const toggleFav = useCallback((med: Medication) => {

   104→   104→      removeFavorite(med.id);

   106→   106→      addFavorite({
   107→   107→        medicationId: med.id,
   108→   108→        name: med.name,
   109→   109→        genericName: med.genericName,
   110→   110→        category: med.categoryId,
   111→   111→        species: med.species,
   112→   112→      });
   113→   113→    }

   115→   115→  }, []);
   116→   116→

   122→   122→      meds = meds.filter(
   123→   123→        (m) =>
   124→   124→          m.name.toLowerCase().includes(q) ||
   125→   125→          m.genericName.toLowerCase().includes(q) ||
   126→   126→          (m.brandNames && m.brandNames.some((b) => b.toLowerCase().includes(q)))
   127→   127→      );
   128→   128→    }
   129→   129→    return meds;
   130→   130→  }, [selectedCategory, searchQuery, favTick]);
   131→   131→
   132→   132→  const canCalculate = selectedMedication && weight && parseFloat(weight) > 0;
   133→   133→
   134→   134→  const handleCalculate = async () => {
   135→   135→    if (!selectedMedication || !weight) return;
   136→   136→    setLoading(true);
   137→   137→    setError(null);
   138→   138→    setResult(null);
   139→   139→
   140→   140→    try {
   141→   141→      const res = await fetch('/api/calculate-medication', {
   142→   142→        method: 'POST',
   143→   143→        headers: { 'Content-Type': 'application/json' },
   144→   144→        body: JSON.stringify({
   145→   145→          medicationId: selectedMedication.id,
   146→   146→          animalType,
   147→   147→          weight: parseFloat(weight),
   148→   148→          weightUnit,
   149→   149→        }),
   150→   150→      });
   151→   151→
   152→   152→      const data = await res.json();
   153→   153→
   154→   154→      if (!res.ok) {
   155→   155→        setError(data.error || 'Error al calcular la dosis');
   156→   156→        return;
   157→   157→      }
   158→   158→
   159→   159→      setResult(data);
   160→   160→      // Save to history
   161→   161→      const unit = data.calculatedDose.unit.split('/')[0];
   162→   162→      addHistory({
   163→   163→        type: 'medication',
   164→   164→        timestamp: Date.now(),
   165→   165→        summary: `${data.medication.name} | ${animalType} ${data.weightKg}kg | ${data.calculatedDose.recommended}${unit}`,
   166→   166→      });
   167→   167→      // history auto-updates via useSyncExternalStore
   168→   168→    } catch {
   169→   169→      setError('Error de conexión. Intente de nuevo.');
   170→   170→    } finally {
   171→   171→      setLoading(false);
   172→   172→    }
   173→   173→  };
   174→   174→
   175→   175→  const handleReset = () => {
   176→   176→    setSelectedMedication(null);
   177→   177→    setResult(null);
   178→   178→    setError(null);
   179→   179→  };
   180→   180→
   181→   181→  return (
   182→   182→    <div className="space-y-6">
   183→   183→      {/* Recent History */}
   184→   184→      {history.length > 0 && !result && (
   185→   185→        <div className="no-print">
   186→   186→          <div className="flex items-center gap-2 mb-2">
   187→   187→            <Clock size={16} weight="outline" className="text-muted-foreground" />
   188→   188→            <span className="text-sm font-medium text-muted-foreground">Consultas recientes</span>
   189→   189→          </div>
   190→   190→          <div className="flex gap-2 overflow-x-auto pb-1">
   191→   191→            {history.slice(0, 5).map((h, i) => (
   192→   192→              <div
   193→   193→                key={i}
   194→   194→                className="flex-shrink-0 bg-muted/60 rounded-lg px-3 py-1.5 text-xs text-muted-foreground border border-border/50"
   195→   195→              >
   196→   196→                {h.summary}
   197→   197→              </div>
   198→   198→            ))}
   199→   199→          </div>
   200→   200→        </div>
   201→   201→      )}
   202→   202→
   203→   203→      {/* Step 1: Animal Type */}
   204→   204→      <div>
   205→   205→        <StepHeading num={1}>Tipo de Animal</StepHeading>
   206→   206→        <div className="grid grid-cols-2 gap-3">
   207→   207→          {(['perro', 'gato'] as const).map((type) => (
   208→   208→            <button
   209→   209→              key={type}
   210→   210→              onClick={() => { setAnimalType(type); setSelectedMedication(null); setResult(null); }}
   211→   211→              className={`p-4 rounded-xl border-2 transition-all duration-200 flex flex-col items-center gap-2 ${
   212→   212→                animalType === type
   213→   213→                  ? 'border-primary bg-primary/5 shadow-md shadow-primary/10'
   214→   214→                  : 'border-border hover:border-primary/30'
   215→   215→              }`}
   216→   216→            >
   217→   217→              {type === 'perro' ? (
   218→   218→                <span className="text-4xl">🐕</span>
   219→   219→              ) : (
   220→   220→                <Cat size={36} weight="outline" color={animalType === 'gato' ? 'oklch(0.55 0.15 165)' : 'oklch(0.5 0.02 165)'} />
   221→   221→              )}
   222→   222→              <span className={`font-semibold ${animalType === type ? 'text-primary' : 'text-muted-foreground'}`}>
   223→   223→                {type === 'perro' ? 'Perro' : 'Gato'}
   224→   224→              </span>
   225→   225→            </button>
   226→   226→          ))}
   227→   227→        </div>
   228→   228→      </div>
   229→   229→
   230→   230→      {/* Step 2: Weight */}
   231→   231→      <div>
   232→   232→        <StepHeading num={2}>Peso del Animal</StepHeading>
   233→   233→        <div className="flex gap-3 items-center">
   234→   234→          <Input
   235→   235→            type="number"
   236→   236→            placeholder="Ej: 5"
   237→   237→            value={weight}
   238→   238→            onChange={(e) => { setWeight(e.target.value); setResult(null); }}
   239→   239→            min="0.1"
   240→   240→            step="0.1"
   241→   241→            className="flex-1 h-12 text-lg"
   242→   242→          />
   243→   243→          <div className="flex rounded-lg border border-border overflow-hidden">
   244→   244→            {(['kg', 'lb'] as const).map((u) => (
   245→   245→              <button
   246→   246→                key={u}
   247→   247→                onClick={() => setWeightUnit(u)}
   248→   248→                className={`px-4 py-3 text-sm font-semibold transition-colors ${
   249→   249→                  weightUnit === u ? 'bg-primary text-primary-foreground' : 'bg-card hover:bg-muted'
   250→   250→                }`}
   251→   251→              >
   252→   252→                {u}
   253→   253→              </button>

   257→   257→      </div>
   258→   258→
   259→   259→      {/* Step 3: Category Selection */}

   262→   262→          <StepHeading num={3}>Categoría de Medicamento</StepHeading>

   265→   265→            size="sm"

   270→   270→            Favoritos

   273→   273→        <div className="flex gap-2 overflow-x-auto pb-2 -mx-1 px-1 no-print">
   274→   274→          {medicationCategories.map((cat) => (
   275→   275→            <button
   276→   276→              key={cat.id}
   277→   277→              onClick={() => {
   278→   278→                setSelectedCategory(cat.id === selectedCategory ? null : cat.id);
   279→   279→                setSelectedMedication(null);
   280→   280→                setResult(null);
   281→   281→                setSearchQuery('');
   282→   282→              }}
   283→   283→              className={`flex-shrink-0 flex items-center gap-2 px-4 py-2.5 rounded-full text-sm font-medium border transition-all duration-200 whitespace-nowrap ${
   284→   284→                selectedCategory === cat.id
   285→   285→                  ? 'bg-primary text-primary-foreground border-primary shadow-md shadow-primary/20'
   286→   286→                  : 'bg-card border-border hover:border-primary/30 hover:bg-primary/5'
   287→   287→              }`}
   288→   288→            >
   289→   289→              <span>{cat.icon}</span>
   290→   290→              <span>{cat.name}</span>
   291→   291→            </button>
   292→   292→          ))}
   293→   293→        </div>
   294→   294→      </div>
   295→   295→
   296→   296→      {/* Step 4: Medication List with Search */}
   297→   297→      <AnimatePresence mode="wait">
   298→   298→        {selectedCategory && filteredMedications.length > 0 && (
   299→   299→          <motion.div
   300→   300→            key={selectedCategory}
   301→   301→            initial={{ opacity: 0, y: 10 }}
   302→   302→            animate={{ opacity: 1, y: 0 }}
   303→   303→            exit={{ opacity: 0, y: -10 }}
   304→   304→            transition={{ duration: 0.2 }}
   305→   305→          >
   306→   306→            <div className="flex items-center justify-between gap-3 mb-3">
   307→   307→              <StepHeading num={4}>Seleccione un Medicamento</StepHeading>
   308→   308→              <div className="relative w-40 flex-shrink-0 no-print">
   309→   309→                <Search size={14} weight="outline" className="absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground" />
   310→   310→                <Input
   311→   311→                  type="text"
   312→   312→                  placeholder="Buscar..."
   313→   313→                  value={searchQuery}
   314→   314→                  onChange={(e) => setSearchQuery(e.target.value)}
   315→   315→                  className="h-8 text-sm pl-8"
   316→   316→                />
   317→   317→              </div>
   318→   318→            </div>
   319→   319→
   320→   320→            {/* Dose Reference Table - collapsible */}
   321→   321→            <Collapsible open={showRefTable} onOpenChange={setShowRefTable} className="mb-3 no-print">
   322→   322→              <CollapsibleTrigger asChild>
   323→   323→                <Button
   324→   324→                  variant="ghost"
   325→   325→                  size="sm"
   326→   326→                  className="w-full justify-between gap-2 text-xs text-muted-foreground hover:text-foreground h-8 px-2"
   327→   327→                >
   328→   328→                  <span className="flex items-center gap-1.5">
   329→   329→                    <ClipboardList size={14} weight="outline" />
   330→   330→                    Ver tabla de referencia
   331→   331→                  </span>
   332→   332→                  <motion.span
   333→   333→                    animate={{ rotate: showRefTable ? 180 : 0 }}
   334→   334→                    transition={{ duration: 0.2 }}
   335→   335→                  >
   336→   336→                    <ChevronDown size={14} weight="outline" />
   337→   337→                  </motion.span>
   338→   338→                </Button>
   339→   339→              </CollapsibleTrigger>
   340→   340→              <CollapsibleContent>
   341→   341→                <motion.div
   342→   342→                  initial={{ opacity: 0, height: 0 }}
   343→   343→                  animate={{ opacity: 1, height: 'auto' }}
   344→   344→                  exit={{ opacity: 0, height: 0 }}
   345→   345→                  transition={{ duration: 0.2 }}
   346→   346→                >
   347→   347→                  <DoseReferenceTable categoryId={selectedCategory} animalType={animalType} />
   348→   348→                </motion.div>
   349→   349→              </CollapsibleContent>
   350→   350→            </Collapsible>
   351→   351→
   352→   352→            <div className="grid gap-2.5 max-h-[400px] overflow-y-auto pr-1">
   353→   353→              {filteredMedications.map((med) => {
   354→   354→                const isAvailable = med.species.includes(animalType);
   355→   355→                const isSelected = selectedMedication?.id === med.id;
   356→   356→                const medIsFav = isFavorite(med.id);
   357→   357→
   358→   358→                return (
   359→   359→                  <Card
   360→   360→                    key={med.id}
   361→   361→                    className={`med-card cursor-pointer ${isSelected ? 'selected' : ''} ${
   362→   362→                      !isAvailable ? 'opacity-40 pointer-events-none' : ''
   363→   363→                    }`}
   364→   364→                    onClick={() => {
   365→   365→                      if (!isAvailable) return;
   366→   366→                      setSelectedMedication(isSelected ? null : med);
   367→   367→                      setResult(null);
   368→   368→                    }}
   369→   369→                  >
   370→   370→                    <CardContent className="p-3.5">
   371→   371→                      <div className="flex items-start justify-between gap-2">
   372→   372→                        <div className="flex-1 min-w-0">
   373→   373→                          <div className="flex items-center gap-2 flex-wrap">
   374→   374→                            <h4 className="font-bold text-sm">{med.name}</h4>
   375→   375→                            {med.species.map((s) => (
   376→   376→                              <Badge
   377→   377→                                key={s}
   378→   378→                                variant={s === animalType ? 'default' : 'secondary'}
   379→   379→                                className="text-[10px] px-1.5 py-0"
   380→   380→                              >
   381→   381→                                {s === 'perro' ? '🐕' : '🐈'} {s}
   382→   382→                              </Badge>
   383→   383→                            ))}
   384→   384→                          </div>
   385→   385→                          <p className="text-xs text-muted-foreground mt-0.5">{med.genericName}</p>
   386→   386→                          <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   387→   387→                            <Badge variant="outline" className="text-[10px] font-semibold text-primary border-primary/30">

   394→   394→                                {r}

   396→   396→                            ))}
   397→   397→                          </div>
   398→   398→                        </div>
   399→   399→                        <div className="flex items-center gap-1 flex-shrink-0 mt-1">
   400→   400→                          <button
   401→   401→                            onClick={(e) => {
   402→   402→                              e.stopPropagation();
   403→   403→                              toggleFav(med);

   405→   405→                            className={`p-1 rounded-md transition-all hover:bg-amber-100 dark:hover:bg-amber-900/30 ${
   406→   406→                              medIsFav ? 'text-amber-500' : 'text-muted-foreground/30 hover:text-amber-400'

   410→   410→                            <Star size={16} weight={medIsFav ? 'fill' : 'outline'} />
   411→   411→                          </button>
   412→   412→                          <ChevronRight
   413→   413→                            size={18} weight="outline"
   414→   414→                            className={`transition-all ${isSelected ? 'text-primary rotate-90' : 'text-muted-foreground/40'}`}
   415→   415→                          />
   416→   416→                        </div>
   417→   417→                      </div>
   418→   418→                    </CardContent>
   419→   419→                  </Card>
   420→   420→                );
   421→   421→              })}
   422→   422→            </div>
   423→   423→          </motion.div>
   424→   424→        )}
   425→   425→        {selectedCategory && filteredMedications.length === 0 && (
   426→   426→          <motion.div
   427→   427→            key="empty"
   428→   428→            initial={{ opacity: 0 }} animate={{ opacity: 1 }}
   429→   429→            className="text-center py-8 text-muted-foreground"

   431→   431→            <Search size={32} weight="outline" className="mx-auto mb-2 opacity-30" />

   435→   435→      </AnimatePresence>
   436→   436→
   437→   437→      {/* Calculate Button */}
   438→   438→      <AnimatePresence>
   439→   439→        {selectedMedication && (
   440→   440→          <motion.div

   461→   461→            </Button>
   462→   462→            <p className="text-sm text-muted-foreground">
   463→   463→              {selectedMedication.name} — {selectedMedication.genericName}
