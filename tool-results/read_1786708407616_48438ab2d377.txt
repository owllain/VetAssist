   116→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   117→  const ref = useRef<HTMLSpanElement>(null);
   118→  const isInView = useInView(ref, { once: true });
   119→  const [count, setCount] = useState(0);
   120→
   121→  useEffect(() => {
   122→    if (!isInView) return;
   123→    let start = 0;
   124→    const end = target;
   125→    const duration = 1500;
   126→    const stepTime = 30;
   127→    const steps = duration / stepTime;
   128→    const increment = end / steps;
   129→
   130→    const timer = setInterval(() => {
   131→      start += increment;
   132→      if (start >= end) {
   133→        setCount(end);
   134→        clearInterval(timer);
   135→      } else {
   136→        setCount(Math.floor(start));
   137→      }
   138→    }, stepTime);
   139→    return () => clearInterval(timer);
   140→  }, [isInView, target]);
   141→
   142→  return <span ref={ref}>{count}{suffix}</span>;
   143→}
