   124→   124→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   125→   125→  const ref = useRef<HTMLSpanElement>(null);
   126→   126→  const isInView = useInView(ref, { once: true });
   127→   127→  const [count, setCount] = useState(0);
   128→   128→
   129→   129→  useEffect(() => {
   130→   130→    if (!isInView) return;
   131→   131→    let start = 0;
   132→   132→    const end = target;
   133→   133→    const duration = 1500;
   134→   134→    const stepTime = 30;
   135→   135→    const steps = duration / stepTime;
   136→   136→    const increment = end / steps;
   137→   137→
   138→   138→    const timer = setInterval(() => {
   139→   139→      start += increment;
   140→   140→      if (start >= end) {
   141→   141→        setCount(end);
   142→   142→        clearInterval(timer);
   143→   143→      } else {
   144→   144→        setCount(Math.floor(start));
   145→   145→      }
   146→   146→    }, stepTime);
   147→   147→    return () => clearInterval(timer);
   148→   148→  }, [isInView, target]);
   149→   149→
   150→   150→  return <span ref={ref}>{count}{suffix}</span>;
   151→   151→}
