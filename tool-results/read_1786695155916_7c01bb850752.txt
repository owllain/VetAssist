    44→interface CalcResult {
    45→  medication: {
    46→    id: string;
    47→    name: string;
    48→    genericName: string;
    49→    category: string;
    50→    brandNames: string[];

    53→    min: number;

    63→
    64→interface HistoryEntry {
    65→  type: 'medication' | 'free' | 'food';
    66→  timestamp: number;
    67→  summary: string;
    68→}
    69→
    70→const STORAGE_KEY = 'vetcalc-history';
    71→
    72→function loadHistory(): HistoryEntry[] {
    73→  try { return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]'); } catch { return []; }
    74→}
    75→
    76→function saveHistory(entry: HistoryEntry) {
    77→  const history = loadHistory();
    78→  history.unshift(entry);
    79→  if (history.length > 20) history.pop();
    80→  localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
    81→}
    82→
    83→function StepHeading({ num, children }: { num: number; children: React.ReactNode }) {
    84→  return (
    85→    <h3 className="text-base font-semibold mb-3 flex items-center gap-2.5">
    86→      <span className="step-number">{num}</span>
    87→      {children}
    88→    </h3>
    89→  );
    90→}

   187→    }
   188→  };

   192→    setResult(null);
   193→    setError(null);
   194→  };

   325→                <Input

   342→                >

   345→                    Ver tabla de referencia
   346→                  </span>
   347→                  <motion.span

   349→                    transition={{ duration: 0.2 }}
   350→                  >
   351→                    <ChevronDown size={14} weight="outline" />
   352→                  </motion.span>
   353→                </Button>
   354→              </CollapsibleTrigger>
   355→              <CollapsibleContent>

   358→                  animate={{ opacity: 1, height: 'auto' }}
   359→                  exit={{ opacity: 0, height: 0 }}

   364→              </CollapsibleContent>
   365→            </Collapsible>
   366→
   367→            <div className="grid gap-2.5 max-h-[400px] overflow-y-auto pr-1">
   368→              {filteredMedications.map((med) => {
   369→                const isAvailable = med.species.includes(animalType);
   370→                const isSelected = selectedMedication?.id === med.id;
   371→                const medIsFav = isFavorite(med.id);
   372→
   373→                return (
   374→                  <Card
   375→                    key={med.id}
   376→                    className={`med-card cursor-pointer ${isSelected ? 'selected' : ''} ${
   377→                      !isAvailable ? 'opacity-40 pointer-events-none' : ''

   379→                    onClick={() => {
   380→                      if (!isAvailable) return;
   381→                      setSelectedMedication(isSelected ? null : med);
   382→                      setResult(null);

   384→                  >
   385→                    <CardContent className="p-3.5">

   390→                            {med.species.map((s) => (
   391→                              <Badge

   393→                                variant={s === animalType ? 'default' : 'secondary'}

   395→                              >
   396→                                {s === 'perro' ? '🐕' : '🐈'} {s}
   397→                              </Badge>
   398→                            ))}
   399→                          </div>
   400→                          <p className="text-xs text-muted-foreground mt-0.5">{med.genericName}</p>
   401→                          <div className="flex items-center gap-1.5 mt-2 flex-wrap">
   402→                            <Badge variant="outline" className="text-[10px] font-semibold text-primary border-primary/30">

   431→                        </div>
   432→                      </div>
   433→                    </CardContent>

   437→            </div>
   438→          </motion.div>
   439→        )}
   440→        {selectedCategory && filteredMedications.length === 0 && (
   441→          <motion.div
   442→            key="empty"
   443→            initial={{ opacity: 0 }} animate={{ opacity: 1 }}

   496→      {/* Result */}

   503→            transition={{ duration: 0.3 }}
